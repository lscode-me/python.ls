Q19

Windowing: What does this coroutine do?

@coroutine
def sliding_window(size, target):
    window = []
    while True:
        item = yield
        window.append(item)
        if len(window) > size:
            window.pop(0)
        if len(window) == size:
            target.send(list(window))
